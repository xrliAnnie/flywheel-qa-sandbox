        fatalx("xmalloc: allocating %zu bytes: %s",

xcalloc:

## log.c @ tag 3.5a — fatal()/fatalx() 的终止方式与无日志行为

fatal():
    __dead void
    fatal(const char *msg, ...)
    {
        char     tmp[256];
        va_list  ap;

        if (snprintf(tmp, sizeof tmp, "fatal: %s: ", strerror(errno)) < 0)
            exit(1);

        va_start(ap, msg);
        log_vwrite(msg, ap, tmp);
        va_end(ap);

        exit(1);
    }

fatalx():
    __dead void
    fatalx(const char *msg, ...)
    {
        va_list  ap;

        va_start(ap, msg);
        log_vwrite(msg, ap, "fatal: ");
        va_end(ap);

        exit(1);
    }

终止方式：两者均以 exit(1) 结束——干净退出，非信号、非 abort（因此不产生 .ips 崩溃报告）。
无日志行为：log_level=0（默认，未加 -v）时 log_file 为 NULL；log_vwrite 开头即
    if (log_file == NULL) return;
即默认配置下 fatal/fatalx 什么都不写就退出——「零痕迹死亡」的直接机制依据。
