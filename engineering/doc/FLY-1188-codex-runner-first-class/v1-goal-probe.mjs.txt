// FLY-1188 M4 V1 behavior probe — does thread/goal/set on the remote-control
// daemon actually drive autonomous multi-turn continuation + a complete
// notification? Isolated CODEX_HOME; tiny verifiable objective.
import { spawn } from "node:child_process";
import { existsSync, mkdirSync, readFileSync, readdirSync } from "node:fs";
import { createRequire } from "node:module";
import { join } from "node:path";

const require = createRequire(
  "/Users/xiaorongli/Dev/flywheel-FLY-1188/packages/teamlead/package.json",
);
const WebSocket = require("ws");

const SCRATCH = process.argv[2];
const CODEX_BIN = "/Users/xiaorongli/.codex-infra-bot/packages/standalone/current/codex";
const HOME_DIR = join(SCRATCH, "v1-home");
const WORKDIR = join(SCRATCH, "v1-workdir");
const SOCK_DIR = join(HOME_DIR, "app-server-control");
const SOCK = join(SOCK_DIR, "app-server-control.sock");
mkdirSync(SOCK_DIR, { recursive: true });
mkdirSync(WORKDIR, { recursive: true });

const log = (m) => console.log(`[probe ${new Date().toISOString().slice(11, 19)}] ${m}`);

// 1. daemon (isolated CODEX_HOME)
const daemon = spawn(CODEX_BIN, ["app-server", "--remote-control", "--listen", `unix://${SOCK}`], {
  env: { ...process.env, CODEX_HOME: HOME_DIR },
  stdio: ["ignore", "pipe", "pipe"],
});
daemon.stdout.on("data", (d) => process.stdout.write(`[daemon] ${d}`));
daemon.stderr.on("data", (d) => process.stdout.write(`[daemon!] ${d}`));
daemon.on("exit", (c) => log(`daemon exited ${c}`));

// wait for socket
for (let i = 0; i < 60; i++) {
  if (existsSync(SOCK)) break;
  await new Promise((r) => setTimeout(r, 500));
}
if (!existsSync(SOCK)) {
  log("FAIL: daemon socket never appeared");
  daemon.kill("SIGKILL");
  process.exit(1);
}
log(`daemon socket up: ${SOCK}`);

// 2. WS client (perMessageDeflate MUST be off — daemon-ws precedent)
const ws = new WebSocket(`ws+unix://${SOCK}:/`, { perMessageDeflate: false });
await new Promise((res, rej) => {
  ws.on("open", res);
  ws.on("error", rej);
});
log("ws open");

let nextId = 1;
const pending = new Map();
const turnEvents = [];
let goalNotifications = [];
let turnsStarted = 0;
let turnsCompleted = 0;

ws.on("message", (raw) => {
  let msg;
  try {
    msg = JSON.parse(raw.toString());
  } catch {
    return;
  }
  if (msg.id !== undefined && pending.has(msg.id)) {
    pending.get(msg.id)(msg);
    pending.delete(msg.id);
    return;
  }
  const method = msg.method ?? "";
  if (method === "turn/started") {
    turnsStarted += 1;
    log(`>> turn ${turnsStarted} STARTED`);
  } else if (method === "turn/completed" || method === "turn/complete") {
    turnsCompleted += 1;
    log(`>> turn COMPLETED (total ${turnsCompleted})`);
  } else if (method.includes("goal")) {
    goalNotifications.push(msg.params);
    const st = msg.params?.goal?.status ?? "(cleared)";
    log(`>> GOAL notification: status=${st} tokensUsed=${msg.params?.goal?.tokensUsed ?? "?"} turnId=${msg.params?.turnId ?? "-"}`);
  } else if (method === "error" || method.includes("error")) {
    log(`>> error notif: ${JSON.stringify(msg.params).slice(0, 200)}`);
  } else if (method) {
    turnEvents.push(method);
  }
});

function rpc(method, params, timeoutMs = 30000) {
  const id = nextId++;
  ws.send(JSON.stringify({ jsonrpc: "2.0", id, method, params }));
  return new Promise((resolve, reject) => {
    const t = setTimeout(() => {
      pending.delete(id);
      reject(new Error(`rpc ${method} timeout`));
    }, timeoutMs);
    pending.set(id, (msg) => {
      clearTimeout(t);
      if (msg.error) reject(new Error(`rpc ${method}: ${JSON.stringify(msg.error)}`));
      else resolve(msg.result);
    });
  });
}

// 3. initialize handshake
const init = await rpc("initialize", {
  clientInfo: { name: "fly1188-v1-probe", title: "FLY-1188 V1 probe", version: "0.0.1" },
  capabilities: {},
});
ws.send(JSON.stringify({ jsonrpc: "2.0", method: "initialized", params: {} }));
log(`initialized (server: ${JSON.stringify(init).slice(0, 120)})`);

// 4. thread/start
const thread = await rpc("thread/start", {
  cwd: WORKDIR,
  approvalPolicy: "never",
  sandbox: "workspace-write",
});
const threadId = thread?.thread?.id ?? thread?.threadId ?? thread?.id;
log(`thread started: ${threadId} (raw keys: ${Object.keys(thread ?? {})})`);

// 5. goal set BEFORE any turn — the V1 question is whether the goal machinery
// drives continuation on its own
const OBJECTIVE =
  "In the current working directory, create exactly five files named f1.txt..f5.txt, " +
  "each containing only its own number (1..5) as the sole line. Verify every file's " +
  "content by reading it back. The goal is achieved only when all five files exist " +
  "with correct contents. Work on at most two files per turn.";
await rpc("thread/goal/set", {
  threadId,
  objective: OBJECTIVE,
  status: "active",
  tokenBudget: 200000,
});
log("goal set (active, 200k budget)");
const g0 = await rpc("thread/goal/get", { threadId });
log(`goal readback: ${JSON.stringify(g0).slice(0, 300)}`);

// 6. kick the first turn (mirror of the TUI flow: /goal then work begins)
await rpc("turn/start", {
  threadId,
  input: [{ type: "text", text: "Begin working toward the goal now." }],
}, 60000).catch((e) => log(`turn/start note: ${e.message}`));
log("first turn submitted; observing autonomous continuation for up to 8 minutes…");

// 7. observe
const deadline = Date.now() + 10 * 60000;
let verdict = "TIMEOUT";
let filesDoneAt = null;
while (Date.now() < deadline) {
  await new Promise((r) => setTimeout(r, 5000));
  const files = readdirSync(WORKDIR).filter((f) => /^f[1-5]\.txt$/.test(f));
  if (files.length === 5 && filesDoneAt === null) {
    filesDoneAt = Date.now();
    log("all 5 files present — waiting for goal status=complete notification…");
  }
  const done = goalNotifications.some((g) => g?.goal?.status === "complete");
  if (files.length === 5 && done) {
    verdict = "PASS";
    break;
  }
  // files done but no complete signal within 4 minutes → partial verdict
  if (filesDoneAt !== null && Date.now() - filesDoneAt > 4 * 60000) {
    verdict = "FILES_OK_NO_COMPLETE_SIGNAL";
    break;
  }
  if (daemon.exitCode !== null) {
    verdict = "DAEMON_DIED";
    break;
  }
}

// 8. final goal state + file check
const gEnd = await rpc("thread/goal/get", { threadId }).catch(() => null);
const files = readdirSync(WORKDIR).filter((f) => /^f[1-5]\.txt$/.test(f)).sort();
const contents = files.map((f) => `${f}=${readFileSync(join(WORKDIR, f), "utf8").trim()}`);
log("──── RESULT ────");
log(`verdict: ${verdict}`);
log(`turns started: ${turnsStarted}, completed: ${turnsCompleted}`);
log(`files: ${contents.join(", ") || "(none)"}`);
log(`final goal: ${JSON.stringify(gEnd).slice(0, 400)}`);
log(`goal notifications seen: ${goalNotifications.length}`);
log(`other event methods sample: ${[...new Set(turnEvents)].slice(0, 15).join(", ")}`);

ws.close();
daemon.kill("SIGTERM");
setTimeout(() => daemon.kill("SIGKILL"), 3000).unref();
process.exit(verdict === "PASS" ? 0 : 1);
